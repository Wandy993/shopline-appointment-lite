#!/usr/bin/env bash
set -u

VERSION="0.5.4-hotfix.2"
RELEASE_SLUG="appointment-lite-v0.5.4-hotfix.2-avatar-confirmation-typography"
ZIP_NAME="${RELEASE_SLUG}.zip"
ZIP_PATH="${ZIP_PATH:-$HOME/Downloads/$ZIP_NAME}"
TARGET_DIR="${TARGET_DIR:-$PWD}"
DEPLOY_RAILWAY="${DEPLOY_RAILWAY:-1}"
PUSH_THEME="${PUSH_THEME:-1}"
TEMP_DIR=""

say() { printf '\n[Appointment Lite %s] %s\n' "$VERSION" "$1"; }
fail() {
  printf '\n========================================\n' >&2
  printf 'Appointment Lite %s stopped: %s\n' "$VERSION" "$1" >&2
  printf '========================================\n' >&2
  if [ -n "$TEMP_DIR" ]; then printf 'Temporary files retained at: %s\n' "$TEMP_DIR" >&2; fi
  exit 1
}

[ -f "$ZIP_PATH" ] || fail "ZIP not found: $ZIP_PATH"
TARGET_DIR="$(cd "$TARGET_DIR" && pwd -P)" || fail "Cannot resolve project directory."
[ -d "$TARGET_DIR/.git" ] || fail "Existing Appointment Lite Git repository not found at $TARGET_DIR"

say "1/13 Verify release ZIP"
unzip -t "$ZIP_PATH" >/dev/null || fail "ZIP integrity check failed."

say "2/13 Check Git and sync remote before overlay"
CURRENT_BRANCH="$(git -C "$TARGET_DIR" branch --show-current)"
[ -n "$CURRENT_BRANCH" ] || fail "Detached Git HEAD detected."
if [ -n "$(git -C "$TARGET_DIR" status --porcelain 2>/dev/null)" ]; then
  git -C "$TARGET_DIR" status
  fail "Working tree has uncommitted changes. Commit or stash them before applying v0.5.4-hotfix.2."
fi
if git -C "$TARGET_DIR" remote get-url origin >/dev/null 2>&1; then
  git -C "$TARGET_DIR" fetch origin "$CURRENT_BRANCH" || fail "git fetch failed."
  if git -C "$TARGET_DIR" show-ref --verify --quiet "refs/remotes/origin/$CURRENT_BRANCH"; then
    if git -C "$TARGET_DIR" merge-base --is-ancestor HEAD "origin/$CURRENT_BRANCH" && ! git -C "$TARGET_DIR" merge-base --is-ancestor "origin/$CURRENT_BRANCH" HEAD; then
      git -C "$TARGET_DIR" merge --ff-only "origin/$CURRENT_BRANCH" || fail "Fast-forward failed."
    elif ! git -C "$TARGET_DIR" merge-base --is-ancestor "origin/$CURRENT_BRANCH" HEAD && ! git -C "$TARGET_DIR" merge-base --is-ancestor HEAD "origin/$CURRENT_BRANCH"; then
      fail "Local and origin/$CURRENT_BRANCH have diverged. Resolve Git first; no release files have been changed."
    fi
  fi
fi

say "3/13 Create upgrade backup tag"
BACKUP_TAG="appointment-lite-pre-v0.5.4-hotfix.2-$(date +%Y%m%d-%H%M%S)"
git -C "$TARGET_DIR" tag "$BACKUP_TAG" HEAD || fail "Git backup tag failed."
printf 'Backup tag: %s\n' "$BACKUP_TAG"

say "4/13 Extract release"
TEMP_DIR="$(mktemp -d "${TMPDIR:-/tmp}/appointment-lite-v054h2.XXXXXX")" || fail "Could not create temp directory."
unzip -q "$ZIP_PATH" -d "$TEMP_DIR" || fail "ZIP extraction failed."
SOURCE_DIR="$TEMP_DIR/$RELEASE_SLUG"
[ -f "$SOURCE_DIR/package.json" ] && [ -f "$SOURCE_DIR/src/server.js" ] || fail "Release root is invalid."

say "5/13 Overlay v0.5.4-hotfix.2 source"
rsync -a \
  --exclude='.git/' \
  --exclude='.env' \
  --exclude='theme-app-extension/' \
  --exclude='node_modules/' \
  --exclude='dist/' \
  --exclude='.shopline-cli.yml' \
  "$SOURCE_DIR/" "$TARGET_DIR/" || fail "rsync overlay failed."

say "6/13 Install locked dependencies"
(cd "$TARGET_DIR" && npm ci) || fail "npm ci failed."

say "7/13 Run full tests"
(cd "$TARGET_DIR" && npm test) || fail "npm test failed."

say "8/13 Run JavaScript checks"
(cd "$TARGET_DIR" && npm run check) || fail "npm run check failed."

say "9/13 Verify portrait, confirmation and typography fixes"
ACTUAL_VERSION="$(cd "$TARGET_DIR" && node -p "require('./package.json').version")" || fail "Cannot read package version."
[ "$ACTUAL_VERSION" = "$VERSION" ] || fail "Expected package version $VERSION, got $ACTUAL_VERSION."
grep -q 'publicStaffAvatar' "$TARGET_DIR/src/services/staffing.js" || fail "Embedded storefront avatar resolver is missing."
grep -q 'flex:0 0 auto!important' "$TARGET_DIR/theme-extension-source/public/appointment-lite.css" || fail "Compact confirmation flex reset is missing."
grep -q 'BlinkMacSystemFont' "$TARGET_DIR/theme-extension-source/public/appointment-lite.css" || fail "Unified storefront typography is missing."
[ -f "$TARGET_DIR/public/staff-avatars/staff-1.webp" ] || fail "Bundled AI staff portraits are missing."
[ "$(wc -c < "$TARGET_DIR/public/staff-avatars/staff-1.webp")" -gt 10000 ] || fail "Staff portrait assets are unexpectedly small."
grep -q 'al-rule:${VERSION}' "$TARGET_DIR/theme-extension-source/public/appointment-lite.js" || fail "Versioned storefront rule cache is missing."
grep -q "const VERSION = '0.5.4-hotfix.2'" "$TARGET_DIR/theme-extension-source/public/appointment-lite.js" || fail "Theme asset version is not v0.5.4-hotfix.2."

say "10/13 Sync Theme App Extension source"
if [ -d "$TARGET_DIR/theme-app-extension" ]; then
  mkdir -p "$TARGET_DIR/theme-app-extension/blocks" "$TARGET_DIR/theme-app-extension/public" "$TARGET_DIR/theme-app-extension/i18n"
  rsync -a --delete "$TARGET_DIR/theme-extension-source/blocks/" "$TARGET_DIR/theme-app-extension/blocks/" || fail "Theme blocks sync failed."
  rsync -a --delete "$TARGET_DIR/theme-extension-source/public/" "$TARGET_DIR/theme-app-extension/public/" || fail "Theme public assets sync failed."
  rsync -a --delete "$TARGET_DIR/theme-extension-source/i18n/" "$TARGET_DIR/theme-app-extension/i18n/" || fail "Theme i18n sync failed."
else
  printf 'theme-app-extension is not present; Theme Extension push will be skipped.\n'
  PUSH_THEME=0
fi

say "11/13 Commit and push v0.5.4-hotfix.2"
git -C "$TARGET_DIR" add -A -- . ':!theme-app-extension' 2>/dev/null || git -C "$TARGET_DIR" add -A -- . || fail "git add failed."
git -C "$TARGET_DIR" reset -- theme-app-extension >/dev/null 2>&1 || true
if ! git -C "$TARGET_DIR" diff --cached --quiet; then
  git -C "$TARGET_DIR" commit -m "fix: Appointment Lite v0.5.4-hotfix.2 avatar confirmation typography" || fail "Release commit failed."
fi
if git -C "$TARGET_DIR" remote get-url origin >/dev/null 2>&1; then
  git -C "$TARGET_DIR" push origin "$CURRENT_BRANCH" || fail "git push failed."
fi

say "12/13 Queue Railway deployment (detached)"
if [ "$DEPLOY_RAILWAY" = "1" ]; then
  command -v railway >/dev/null 2>&1 || fail "Railway CLI not found."
  (cd "$TARGET_DIR" && railway status) || fail "Railway project is not linked correctly."
  if railway up --help 2>&1 | grep -q -- '--detach'; then
    (cd "$TARGET_DIR" && railway up --detach) || fail "Railway upload/queue failed."
    printf 'Railway build queued. Continuing immediately to Theme Extension push.\n'
  else
    fail "This Railway CLI does not support --detach. Run 'railway upgrade' once, then rerun this command."
  fi
else
  printf 'Railway deployment skipped because DEPLOY_RAILWAY=%s.\n' "$DEPLOY_RAILWAY"
fi

say "13/13 Push SHOPLINE Theme App Extension"
if [ "$PUSH_THEME" = "1" ]; then
  command -v sl >/dev/null 2>&1 || fail "SHOPLINE CLI not found."
  (cd "$TARGET_DIR/theme-app-extension" && sl extension push) || fail "sl extension push failed."
else
  printf 'Theme push skipped because PUSH_THEME=%s.\n' "$PUSH_THEME"
fi

printf '\n========================================\n'
printf 'Appointment Lite v0.5.4-hotfix.2 complete\n'
printf '========================================\n'
printf 'Backup tag: %s\n' "$BACKUP_TAG"
git -C "$TARGET_DIR" status
